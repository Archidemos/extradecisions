    id = ARESP.100
    hide_window = yes
    is_triggered_only = yes
    has_dlc = "Holy Fury"
    trigger = {
    	has_character_modifier = reclamation_preparation_duration
		not_reclamation_check_trigger = yes
    	has_global_flag = active_reclamation
    }
	
    immediate = {
    	random_list = {
    		# Gain troops
    		5 = {
    			character_event = { id = ARESP.101 }
    		}
    		# Ask the Pope for money / help
    		4 = {
    			trigger = {
    				NOR = {
    					has_character_modifier = reconquista_pope_aid_event_cooldown
    					has_character_flag = reconquista_adventurer
    				}
    				rightful_religious_head_scope = {
    					always = yes
    				}
    			}
    			character_event = { id = ARESP.102 }
    		}
    		# Come across a great Commander to lead your troops
    		2 = {
    			trigger = {
    				NOT = {
    					has_character_modifier = reconquista_commander_event_cooldown
    				}
    			}
    			character_event = { id = ARESP.103 }
    		}
    		# Boats
    		2 = {
    			trigger = {
    				NOT = {
    					has_character_modifier = reconquista_boat_event_cooldown
    				}
    			}
    			character_event = { id = ARESP.104 }
    		}
    		# Boats
    		2 = {
    			trigger = {
    				NOT = {
    					has_character_modifier = reconquista_ally_event_cooldown
    				}
    			}
    			character_event = { id = ARESP.104 }
    		}
    	}
    }

    after = {
    	character_event = { id = ARESP.100 days = 25 }
    }
