namespace = ROMANAUR

character_event = {
	id = ROMANAUR.98
    hide_window = yes
    is_triggered_only = yes
	
    trigger = {
		start_date == 270.5.1
		is_save_game = no
		NOT = { has_global_flag = west_roman_player_event_fired }
    }
	
	immediate = {
		set_global_flag = west_roman_player_event_fired
		set_global_flag = roman_america_part_1_triggered
		
		any_player = { narrative_event = { id = ROMANAUR.99 days = 5 } }
		e_roman_empire = { save_event_target_as = character_rome_crisis }
	}
}

		prestige = -1000

			any_de_jure_vassal = {
				limit = {
					has_landed_title = k_rome_dacia
				}
				set_defacto_liege = THIS
			}
		}
		hidden_tooltip = {
			k_rome_dacia = {
				holder_scope = {
					character_event = { id = ROMANAUR.102 }
					save_event_target_as = dacia_ruler_270
					set_truce = {
						who = ROOT
						years = 3
					}
				}
			}
			ROOT = {
				set_truce = {
					who = event_target:dacia_ruler_270
					years = 3
				}
			}
		}
		remove_claim = k_rome_dacia
	}
}
