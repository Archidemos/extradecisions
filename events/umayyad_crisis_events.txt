character_event = {
	id = UMM.10
    hide_window = yes
    is_triggered_only = yes
	
    trigger = {
		NOR = {
			start_date == 276.5.1
			start_date == 400.1.1
			start_date == 476.1.1
		}
		start_date <= 936.8.7
		is_save_game = no
		NOT = { has_global_flag = umayyad_player_event_fired }
    }
	
	immediate = {
		set_global_flag = umayyad_player_event_fired
		any_player = { narrative_event = { id = UMM.6 days = 5 } }
		k_andalusia = {
			save_event_target_as = character_umayyad_crisis
		}
	}
}

All hope is lost in a united Islamic Spain and will leave the muslim realms exposed to possible Christian expansion."	
    is_triggered_only = yes
	
	portrait = character_umayyad_crisis
	border = GFX_event_narrative_frame_war
	picture = GFX_evt_mecca_falls_soa
	
    trigger = {
		dynasty = 597
		OR = {
			has_landed_title = k_andalusia
			has_landed_title = e_cordoba
		}
		religion_group = muslim
		start_date <= 1000.1.1
		date >= 1000.1.1
		OR = {
			has_global_flag = umayyad_crisis_1
			has_global_flag = umayyad_crisis_2
			has_global_flag = umayyad_crisis_3 
		}
		NOT = { has_global_flag = umayyad_crisis_done }
    }
	
	option = {
		name = "Ya Allah.."
		set_global_flag = umayyad_crisis_done
		custom_tooltip = {
			text = "Because you were not able to resolve all the issues within the Umayyad Caliphate of Cordoba before the year 1000. All of your higher tier titles will get destroyed."
			hidden_tooltip = {
				ROOT = {
					remove_character_modifier = umayyad_curse_1
					remove_character_modifier = umayyad_curse_2
					remove_character_modifier = umayyad_curse_3
					any_demesne_title = {
						limit = { 
							OR = {
								tier = KING
								tier = EMPEROR
							}
						}
						unsafe_destroy_landed_title = THIS
					}
				}
			}
		}
		any_player = {
			limit = { NOT = { character = ROOT } }
			narrative_event = { id = UMM.31 days = 1 } 
		}
	}
}
